<script>
  (function () {
    var canvas = document.getElementById("matrix-rain");
    if (!canvas) return;
    if (window.matchMedia && window.matchMedia("(prefers-reduced-motion: reduce)").matches) return;

    var ctx = canvas.getContext("2d");
    var glyphs = "アイウエオカキクケコサシスセソタチツテトナニヌネノハヒフヘホ0123456789Z:=*+-<>";
    var fontSize = 16;
    var drops = [];

    function size() {
      canvas.width = window.innerWidth;
      canvas.height = window.innerHeight;
      var columns = Math.ceil(canvas.width / fontSize);
      while (drops.length < columns) {
        drops.push(Math.floor(Math.random() * -60));
      }
      drops.length = columns;
      ctx.fillStyle = "#0c1310";
      ctx.fillRect(0, 0, canvas.width, canvas.height);
    }

    size();
    window.addEventListener("resize", size);

    setInterval(function () {
      /* fade previous frame toward the page background for the trail effect */
      ctx.fillStyle = "rgba(12, 19, 16, 0.08)";
      ctx.fillRect(0, 0, canvas.width, canvas.height);
      ctx.font = fontSize + "px monospace";
      for (var i = 0; i < drops.length; i++) {
        var text = glyphs.charAt(Math.floor(Math.random() * glyphs.length));
        var y = drops[i] * fontSize;
        ctx.fillStyle = "#00ff41";
        ctx.fillText(text, i * fontSize, y);
        if (y > canvas.height && Math.random() > 0.975) {
          drops[i] = 0;
        }
        drops[i]++;
      }
    }, 50);
  })();
</script>
